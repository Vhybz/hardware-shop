2. STEP 1: LARAVEL 11 PROJECT & POSTGRESQL SETUP
================================================================================

1. Create a new Laravel 11 API project:
   composer create-project laravel/laravel:^11.0 laravel-pos-api
   cd laravel-pos-api

2. Install Laravel Sanctum for API token authentication:
   composer require laravel/sanctum
   php artisan vendor:publish --provider="Laravel\Sanctum\SanctumServiceProvider"

3. Update your `.env` file for PostgreSQL connection:

   APP_NAME="Adom Mmoroso ENT. API"
   APP_ENV=production
   APP_KEY=base64:...
   APP_DEBUG=false
   APP_URL=https://api.adomhardware.com

   DB_CONNECTION=pgsql
   DB_HOST=127.0.0.1
   DB_PORT=5432
   DB_DATABASE=adom_pos_db
   DB_USERNAME=postgres
   DB_PASSWORD=your_secure_password

   SESSION_DRIVER=database
   SANCTUM_STATEFUL_DOMAINS=adomhardware.com

-- CONTROLLER: `app/Http/Controllers/Api/AuthController.php`
--------------------------------------------------------------------------------
namespace App\Http\Controllers\Api;

use App\Http\Controllers\Controller;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\Auth;
use App\Models\User;

class AuthController extends Controller {

    public function login(Request $request) {
        $credentials = $request->validate([
            'email' => 'required|email',
            'password' => 'required',
        ]);

        if (!Auth::attempt($credentials)) {
            return response()->json([
                'status' => 'error',
                'message' => 'Invalid email address or password'
            ], 401);
        }

        $user = Auth::user();

        if ($user->status !== 'approved') {
            return response()->json([
                'status' => 'error',
                'message' => 'Your staff account is pending approval.'
            ], 403);
        }

        // Generate Sanctum Bearer Token
        $token = $user->createToken('pos_device_token')->plainTextToken;

        return response()->json([
            'status' => 'success',
            'token' => $token,
            'user' => $user,
        ]);
    }

    public function me(Request $request) {
        return response()->json([
            'status' => 'success',
            'user' => $request->user(),
        ]);
    }
}

-- CONTROLLER: `app/Http/Controllers/Api/SaleController.php`
--------------------------------------------------------------------------------
namespace App\Http\Controllers\Api;

use App\Http\Controllers\Controller;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\DB;
use App\Models\Sale;
use App\Models\Product;

class SaleController extends Controller {

    public function store(Request $request) {
        $validated = $request->validate([
            'id' => 'required|string',
            'branch_code' => 'required|string',
            'cashier_id' => 'required|string',
            'cashier_name' => 'required|string',
            'total_amount' => 'required|numeric',
            'amount_paid' => 'required|numeric',
            'balance' => 'required|numeric',
            'status' => 'required|string',
            'items' => 'required|array',
            'payments' => 'required|array',
        ]);

        // Wrap in PostgreSQL DB Transaction for 100% Concurrency Safety
        return DB::transaction(function () use ($validated, $request) {

            // 1. Create Invoice Record
            $sale = Sale::create([
                'id' => $validated['id'],
                'branch_code' => $validated['branch_code'],
                'cashier_id' => $validated['cashier_id'],
                'cashier_name' => $validated['cashier_name'],
                'customer_id' => $request->input('customer_id'),
                'customer_name' => $request->input('customer_name'),
                'customer_phone' => $request->input('customer_phone'),
                'total_amount' => $validated['total_amount'],
                'amount_paid' => $validated['amount_paid'],
                'balance' => $validated['balance'],
                'discount_amount' => $request->input('discount_amount', 0),
                'status' => $validated['status'],
                'items' => $validated['items'],
                'payments' => $validated['payments'],
                'timestamp' => now(),
            ]);

            // 2. Atomic Stock Decrement per Product
            foreach ($validated['items'] as $item) {
                Product::where('id', $item['product_id'])
                    ->decrement('stock_quantity', $item['quantity']);
            }

            return response()->json([
                'status' => 'success',
                'message' => 'Sale recorded successfully',
                'sale' => $sale,
            ], 201);
        });
    }

    public function index(Request $request) {
        $branchCode = $request->query('branch_code');
        $sales = Sale::where('branch_code', $branchCode)
            ->where('is_active', true)
            ->orderBy('timestamp', 'desc')
            ->paginate(50);

        return response()->json([
            'status' => 'success',
            'data' => $sales,
        ]);
    }
}

7. STEP 6: FLUTTER LARAVEL API SERVICE LAYER
================================================================================

-- Create `lib/services/laravel_api_service.dart` in your Flutter POS project:

import 'dart:convert';
import 'package:http/http.dart' as http;
import 'package:hive_flutter/hive_flutter.dart';

class LaravelApiService {
  static const String baseUrl = 'https://api.adomhardware.com/api/v1';

  static String? get _token {
    final box = Hive.box('app_settings');
    return box.get('laravel_token');
  }

  static Map<String, String> get _headers => {
    'Content-Type': 'application/json',
    'Accept': 'application/json',
    if (_token != null) 'Authorization': 'Bearer $_token',
  };

  static Future<Map<String, dynamic>> login(String email, String password) async {
    final response = await http.post(
      Uri.parse('$baseUrl/auth/login'),
      headers: _headers,
      body: jsonEncode({'email': email, 'password': password}),
    );

    final data = jsonDecode(response.body);

    if (response.statusCode == 200 && data['status'] == 'success') {
      final token = data['token'];
      final box = Hive.box('app_settings');
      await box.put('laravel_token', token);
      return data;
    }
    throw Exception(data['message'] ?? 'Login failed');
  }

  static Future<List<dynamic>> fetchProducts(String branchCode) async {
    final response = await http.get(
      Uri.parse('$baseUrl/products?branch_code=$branchCode'),
      headers: _headers,
    );

    if (response.statusCode == 200) {
      final data = jsonDecode(response.body);
      return data['data'];
    }
    throw Exception('Failed to load products');
  }

  static Future<bool> submitSale(Map<String, dynamic> saleJson) async {
    final response = await http.post(
      Uri.parse('$baseUrl/sales'),
      headers: _headers,
      body: jsonEncode(saleJson),
    );

    return response.statusCode == 201;
  }
}

8. STEP 7: CLOUD PRODUCTION HOSTING GUIDE
================================================================================

OPTION A: MANAGED CLOUD PLATFORM (Render.com / Railway.app)
--------------------------------------------------------------------------------
1. Create a Managed PostgreSQL Database on Render/Railway.
2. Push your Laravel project repository to GitHub.
3. On Render, click "New Web Service" -> Connect GitHub repo.
4. Set Build Command: `composer install --no-dev && php artisan config:cache`
5. Set Start Command: `php artisan migrate --force && php artisan serve --host=0.0.0.0 --port=8080`
6. Add Environment Variables in Dashboard: `DB_CONNECTION`, `DB_HOST`, `DB_DATABASE`, `DB_USERNAME`, `DB_PASSWORD`.


OPTION B: PRODUCTION UBUNTU VPS (DigitalOcean / Linode / Hetzner)
--------------------------------------------------------------------------------
1. Connect to VPS via SSH:
   ssh root@your_vps_ip

2. Install Nginx, PHP 8.3-FPM, PostgreSQL 16:
   sudo apt update && sudo apt upgrade -y
   sudo apt install nginx php8.3-fpm php8.3-pgsql php8.3-mbstring php8.3-xml php8.3-bcmath postgresql postgresql-contrib git unzip certbot python3-certbot-nginx -y

3. Configure PostgreSQL Database:
   sudo -u postgres psql
   CREATE DATABASE adom_pos_db;
   CREATE USER adom_user WITH PASSWORD 'YourSecretPass123!';
   GRANT ALL PRIVILEGES ON DATABASE adom_pos_db TO adom_user;
   \q

4. Clone Laravel API Project to `/var/www/adom-api`:
   cd /var/www
   git clone https://github.com/your-repo/laravel-pos-api.git adom-api
   cd adom-api
   composer install --no-dev --optimize-autoloader
   cp .env.example .env
   php artisan key:generate
   php artisan migrate --force
   sudo chown -R www-data:www-data /var/www/adom-api/storage /var/www/adom-api/bootstrap/cache

5. Configure Nginx Server Block (`/etc/nginx/sites-available/adom-api`):

   server {
       listen 80;
       server_name api.adomhardware.com;
       root /var/www/adom-api/public;

       index index.php;

       location / {
           try_files $uri $uri/ /index.php?$query_string;
       }

       location ~ \.php$ {
           include snippets/fastcgi-php.conf;
           fastcgi_pass unix:/var/run/php/php8.3-fpm.sock;
       }
   }

   sudo ln -s /etc/nginx/sites-available/adom-api /etc/nginx/sites-enabled/
   sudo nginx -t
   sudo systemctl reload nginx

6. Enable Free SSL HTTPS Certificate:
   sudo certbot --nginx -d api.adomhardware.com
